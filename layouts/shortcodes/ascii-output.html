{{/* ascii-output shortcode — render a preformatted ASCII block with status markers colored per audit L5: [OK] green, [X] red, [N/A] gray, [ADV] blue. Also color-codes single-letter "Grade: A|B|C|D|F" lines (audit L6). Usage: {{< ascii-output >}} [OK] check 1 [X] check 2 [N/A] check 3 [ADV] advisory line Grade: B {{< /ascii-output >}} */}} {{- $raw := trim .Inner "\n" -}} {{- $html := htmlEscape $raw -}} {{- $html = replace $html "[OK]" "[OK]" -}} {{- $html = replace $html "[X]" "[X]" -}} {{- $html = replace $html "[N/A]" "[N/A]" -}} {{- $html = replace $html "[ADV]" "[ADV]" -}} {{- /* Boutique redesign: B/C/D were lime/amber/orange tuned for a near-black card background — 1.7:1-3:1 on white, well under WCAG AA. Darkened to the same hue family at a shade that clears 4.5:1 on the new white panel background (verified against #ffffff and #faf7f2 alike). */ -}} {{- $html = replaceRE `Grade:\s+A\b` "Grade: A" $html -}} {{- $html = replaceRE `Grade:\s+B\b` "Grade: B" $html -}} {{- $html = replaceRE `Grade:\s+C\b` "Grade: C" $html -}} {{- $html = replaceRE `Grade:\s+D\b` "Grade: D" $html -}} {{- $html = replaceRE `Grade:\s+F\b` "Grade: F" $html -}} 
{{ $html | safeHTML }}
